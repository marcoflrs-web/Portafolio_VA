import cv2
import pytesseract
import os

# Ruta de Tesseract (ajusta si es diferente)
pytesseract.pytesseract_cmd = r"C:\Users\MATUTINO\AppData\Local\Programs\Tesseract-OCR\tesseract.exe"

# Lista de placas permitidas
placas_ok = ["QKJW88", "GKORH52", "OKO96T", "LRJWS52", "XYZ789","DSZ-999-Z","AAA-000-A"]

# ======= PLACA 1 =======
ruta1 = "placa1.webp"
if os.path.exists(ruta1):
    placa1 = cv2.imread(ruta1)
    if placa1 is not None:
        placa1 = cv2.resize(placa1, (1000, 400))
        gris1 = cv2.cvtColor(placa1, cv2.COLOR_BGR2GRAY)
        binaria1 = cv2.threshold(gris1, 120, 255, cv2.THRESH_BINARY)[1]

        try:
            texto1 = pytesseract.image_to_string(binaria1, lang="eng")  # inglés para evitar error de idioma
        except pytesseract.TesseractNotFoundError:
            print("No se encontró Tesseract, revisa la ruta.")
            texto1 = ""
        texto1 = texto1.strip().replace(" ", "").replace("\n", "")
        print("Placa detectada 1:", texto1)
        cv2.imshow("Placa 1", binaria1)
    else:
        print("No se pudo abrir la imagen placa1.webp")
        texto1 = ""
else:
    print("No existe el archivo placa1.webp en la carpeta del proyecto")
    texto1 = ""

# ======= PLACA 2 =======
ruta2 = "placa2.jfif"
if os.path.exists(ruta2):
    placa2 = cv2.imread(ruta2)
    if placa2 is not None:
        placa2 = cv2.resize(placa2, (1000, 400))
        gris2 = cv2.cvtColor(placa2, cv2.COLOR_BGR2GRAY)
        binaria2 = cv2.threshold(gris2, 120, 255, cv2.THRESH_BINARY)[1]

        try:
            texto2 = pytesseract.image_to_string(binaria2, lang="eng")
        except pytesseract.TesseractNotFoundError:
            print("No se encontró Tesseract, revisa la ruta.")
            texto2 = ""
        texto2 = texto2.strip().replace(" ", "").replace("\n", "")
        print("Placa detectada 2:", texto2)
        cv2.imshow("Placa 2", binaria2)
    else:
        print("No se pudo abrir la imagen placa2.jfif")
        texto2 = ""
else:
    print("No existe el archivo placa2.jfif en la carpeta del proyecto")
    texto2 = ""

cv2.waitKey(0)
cv2.destroyAllWindows()

# ======= VERIFICACIONES =======
placas_leidas = [texto1, texto2]

print("\n--- PLACAS LEÍDAS ---")
print(placas_leidas)

# Comprobar si están en la lista permitida
if texto1 in placas_ok:
    print("La placa 1 tiene acceso permitido.")
elif texto1 == "":
    print("No se detectó texto en la placa 1.")
else:
    print("La placa 1 NO tiene permiso o fue mal leída.")

if texto2 in placas_ok:
    print("La placa 2 tiene acceso permitido.")
elif texto2 == "":
    print("No se detectó texto en la placa 2.")
else:
    print("La placa 2 NO tiene permiso o fue mal leída.")

# ======= PLACAS RESTANTES =======
faltantes = []
if placas_ok[0] not in placas_leidas:
    faltantes.append(placas_ok[0])
if placas_ok[1] not in placas_leidas:
    faltantes.append(placas_ok[1])
if placas_ok[2] not in placas_leidas:
    faltantes.append(placas_ok[2])
if placas_ok[3] not in placas_leidas:
    faltantes.append(placas_ok[3])
if placas_ok[4] not in placas_leidas:
    faltantes.append(placas_ok[4])

print("\n Placas restantes (no escaneadas):")
print(faltantes)

📚 Glosario de Funciones

OpenCV

OpenCV (Open Source Computer Vision Library) es una biblioteca de código abierto que se utiliza principalmente para procesamiento de imágenes y visión por computadora. En Python, se accede a través del módulo llamado cv2.

import

import es una palabra clave (keyword) que se usa para incluir módulos o bibliotecas externas en tu programa.

cv2.imread

Es una función de OpenCV (en el módulo cv2) que se usa para leer una imagen desde un archivo y cargarla en tu programa como una matriz (array de píxeles).

cv2.imshow

Es una función de OpenCV que sirve para mostrar una imagen en una ventana emergente.
